    /**
     * 連絡登録保存処理
     */
    public function save()
    {
        $this->autoRender = false;

        $office_id = $this->Auth->user('office_id');
        $params = $this->getParam('params');
//$this->log($params, 'debug');
        // NULL項目はNULLで埋める
        $params['required_time'] = null;
        $params['menu_id'] = null;
        $params['menu2_id'] = null;
        $params['menu3_id'] = null;
        $params['staff2_id'] = null;
        $params['datetime_disp'] = null;
        $params['datetime_disp'] = null;
        $params['private_expense'] = null;

        $params['id'] = $params['suspended_id'];
        $params['office_id'] = $office_id;
        $params['notices_on'] = str_replace('/', '-', $params['notices_on']);
        $params['suspended_flag'] = 1;
        $this->SuspendedReservation->save($params);

        // 登録の場合は最終挿入ID、更新の場合はsuspended_reservations.idを返却
        $suspended_id = 0;
        if ($params['suspended_id'] == null) {
            $suspended_id = $this->SuspendedReservation->getLastInsertID();    
        } else {
            $suspended_id = $params['suspended_id'];
        }
        
        return json_encode(array('suspended_id' => $suspended_id));
    }
