// ② 本地要上传的文件路径
const filePath = "/Users/user40010308/Documents/test.png"

async function upload() {
  const body = await readFile(filePath) // 读成 Buffer 直接当 body

  const res = await fetch(presignedUrl, {
    method: "PUT",
    headers: {
      "Content-Type": "image/png",
      "x-amz-storage-class": "STANDARD", // 必须,和 SignedHeaders 对应
    },
    body,
  })

  const text = await res.text() // 成功时为空,失败时是 S3 的 XML 错误
  console.log("HTTP", res.status)
  if (!res.ok) {
    console.error("上传失败,S3 返回:\n", text)
    process.exit(1)
  }
  console.log("上传成功 ✅  filePath = club-apotool/ai-phone/868/identity_document.png")
}

upload().catch((err) => {
  console.error("请求异常:", err)
  process.exit(1)
})
