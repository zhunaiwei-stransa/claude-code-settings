    public function exec_cancel()
    {
        $this->autoRender = false;

        if ($this->request->is('ajax')) {

            // パラメータ取得
            $office_id = h($this->Auth->user('office_id'));
            $reservation_id = h($this->request->data('reservation_id'));
            $cancel_type = h($this->request->data('cancel_type'));
            $params = $this->request->data('params');

            // オンライン診療＋クレジットを契約している医院の場合、クレジットカード削除対象の予約かチェック
            $isCreditRev = false;
            $contract = $this->viewVars['my_contract'];
            $contractMdl = ClassRegistry::init('Contract');
            if ($contractMdl->isTelemedRequiredCredit($contract)) {
                $isCreditRev = $this->Reservation->isCreditReservation($office_id, $reservation_id);
            }

            // 予約キャンセル状態再確認、二重キャンセル防止
            $ds = $this->Reservation->getDataSource();
            try {
                $ds->begin();
            
                $reservation = $this->Reservation->find('first', [
                    'conditions' => [
                        'Reservation.id' => $reservation_id,
                        'Reservation.office_id' => $office_id
                    ],
                    'fields' => ['Reservation.deleted_at'],
                    'lock' => true
                ]);
            
                if (empty($reservation['Reservation']['deleted_at'])) {
                    // 予約レコードにcancel_typeをセット
                    $utc = now2utc();
                    $this->Reservation->setCancelType($reservation_id, $cancel_type, $utc);

                    // 患者理由の場合のキャンセル理由を記録
                    if ($this->request->data('params') != null) {

                        // POSTパラメータをクリーニング
                        foreach ($params as $key => $row) {
                            if($key == 'memo') continue; // keep those html tags!
                            $params[$key] = h($row);
                        }

                        // パラメータ追加
                        $params['reservation_id'] = $reservation_id;
                        $params['office_id'] = $office_id;

                        // キャンセル日時
                        $now = time();
                        $params['cancelled_at'] = jst2utc($now);
                        $params['datetime_disp'] = jst2utc($now);

                        // キャンセルモデル
                        $this->Cancel->saveChange($params);

                    }
                    // 予約に結びついていた技工物を開放
                    $this->Material->canselReservationForMaterial($office_id, $reservation_id);
                }

                $ds->commit();
            } catch (Exception $e) {
                $ds->rollback();
                throw new \InternalErrorException();
            }

            // 操作ログを保存
            $reservation = $this->Reservation->findById($reservation_id, null, null, 1);
            $this->ReservationLog->write($reservation, OPELOG_ACTIVITY_CANCELLED, null, $cancel_type == 'office' ? null : 'without_notice');

            // キャンセルした予約の予約日に、同患者に別の予約があるか？
            $check_result = array();
            $check_date = '';
            if(intval($reservation['Reservation']['block']) == 0 && intval($reservation['Reservation']['patient_id']) > 0) {
                $check_date = date('Y-n-j', utc2jst($reservation['Reservation']['starts_at']));
                $check_starts = gmstrftime('%Y-%m-%d %H:%M:%S', strtotime($check_date));
                $check_ends = gmstrftime('%Y-%m-%d %H:%M:%S', strtotime($check_date . ' +1 day'));
                $check_result = $this->Reservation->find(
                    'first',
                    array(
                        'conditions' => array(
                            'Reservation.office_id' => $office_id,
                            'Reservation.patient_id' => $reservation['Reservation']['patient_id'],
                            'Reservation.id !=' => $reservation_id,
                            'Reservation.starts_at >=' => $check_starts,
                            'Reservation.ends_at <' => $check_ends,
                            'Reservation.deleted_at' => null,
                            'Reservation.block' => 0
                        ),
                        'fields' => array('Reservation.id'),
                        'order' => array('Reservation.starts_at ASC, Reservation.id ASC')
                    )
                );
            }

            // キャンセルする予約がオンライン診療でクレジット登録が必須のメニューの場合
            // オンライン診療で１回限りのクレジットカードを選択している場合はクレジットカード情報を削除する
            // ただし、今回の予約以外にクレジットカード登録が必須の予約が残っている場合は削除しない
            // この処理で失敗した場合は、予約のキャンセル自体はエラーにしない、ログに出力する
            if ($isCreditRev) {
                $patient_id = $reservation['Reservation']['patient_id'];    
                // クレジットカードを削除してOKかチェック
                $isDelete = $this->Reservation->isDeleteCredit($office_id, $patient_id, $reservation_id);
                if ($isDelete) {
                    $regiCardMdl = ClassRegistry::init('RegiCard');
                    $cardInfo = $regiCardMdl->getPatientCard($office_id, $patient_id);
                    if (!empty($cardInfo) && !empty($cardInfo['cust_code']) && !empty($cardInfo['is_one_time'])) {
                        $custCode = $cardInfo['cust_code'];
                        if (!$regiCardMdl->deleteCard($office_id, $patient_id, $custCode)) {
                            $regiCardMdl->logPaySbps(sprintf('credit delete error: office[%s] patient[%s]', $office_id, $patient_id));
                        }
                    }
                }
            }

            return json_encode(array(
                'reservation_id' => Hash::get($check_result, 'Reservation.id', ''),
                'target_date' => $check_date
            ));
        }
    }
