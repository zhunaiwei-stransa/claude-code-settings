        function load_operatioonallog_contents() {
            openLoading();

            var url = '/user/reservation_log/search_log';
            if ($('input[name="page"]').val() > 1) {
                url = url + '/page:' + $('input[name="page"]').val();
            }

            $.ajax({
                url: url,
                data: $('#search_log_form').serializeArray(),
                dataType: 'html',
                method: 'get',
            })
            .done(function(data) {
                $('#operatioonallog_contents').html(data);
                closeLoading();
            })
            .fail(function(data) {
                closeLoading();
            });
        }
